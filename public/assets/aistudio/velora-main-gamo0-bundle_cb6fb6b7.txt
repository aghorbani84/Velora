enum class VeloraDestination(
    val route: String,
    val label: String,
    val labelFa: String,
    val icon: ImageVector
) {
    HOME("home", "Home", "خانه", Icons.Default.Home),
    SEARCH("search", "Search", "جستجو", Icons.Default.Search),
    LIBRARY("library", "Library", "کتابخانه", Icons.Default.LibraryMusic),
    MUSIC("music", "Music", "موسیقی", Icons.Default.MusicNote),
    RADIO("radio", "Radio", "رادیو", Icons.Default.Radio),
    ARTISTS("artists", "Artists", "هنرمندان", Icons.Default.Album),
    VIDEO("video", "Video", "ویدیو", Icons.Default.Movie),
    PLAYER("player", "Player & EQ", "پخش و اکولایزر", Icons.Default.GraphicEq),
    SETTINGS("settings", "Settings", "تنظیمات", Icons.Default.Settings);

    companion object {
        fun fromRoute(route: String?): VeloraDestination {
            return entries.firstOrNull { it.route == route } ?: MUSIC
        }

        /**
         * Primary bottom-bar tabs shown on compact screens:
         * - SIMPLE tier: strictly 4 destinations (Music, Video, Player, Settings)
         * - STANDARD & PRO tiers: 5 hybrid bottom tabs (Home, Search, Library [Music/Artists/Radio], Video, Settings)
         */
        fun visibleForTier(tier: ExperienceTier): List<VeloraDestination> {
            return when (tier) {
                ExperienceTier.SIMPLE -> listOf(
                    MUSIC,
                    VIDEO,
                    PLAYER,
                    SETTINGS
                )
                ExperienceTier.STANDARD,
                ExperienceTier.PRO -> listOf(
                    HOME,
                    SEARCH,
                    LIBRARY,
                    VIDEO,
                    SETTINGS
                )
            }
        }

        /**
         * All routes reachable via bottom tabs, segmented Library hub, or desktop sidebar.
         */
        fun allReachableForTier(tier: ExperienceTier): List<VeloraDestination> {
            return when (tier) {
                ExperienceTier.SIMPLE -> visibleForTier(tier)
                ExperienceTier.STANDARD,
                ExperienceTier.PRO -> entries
            }
        }

        fun homeForTier(tier: ExperienceTier): VeloraDestination {
            return if (tier == ExperienceTier.SIMPLE) MUSIC else HOME
        }
    }
}

fun NavHostController.navigateSingleTopTo(destination: VeloraDestination) {
    navigate(destination.route) {
        popUpTo(graph.findStartDestination().id) {
            saveState = true
        }
        launchSingleTop = true
        restoreState = true
    }
}

@Composable
fun VeloraNavGraph(
    navController: NavHostController,
    viewModel: VeloraViewModel,
    uiState: VeloraUiState,
    positionState: PlaybackPositionState = PlaybackPositionState(
        positionMs = uiState.engineState.positionMs,
        durationMs = uiState.engineState.durationMs,
        isPlaying = uiState.engineState.isPlaying
    ),
    startDestination: VeloraDestination = VeloraDestination.MUSIC,
    librarySubTab: LibrarySubTab = LibrarySubTab.PLAYLISTS,
    onSelectLibrarySubTab: (LibrarySubTab) -> Unit = {},
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current

    NavHost(
        navController = navController,
        startDestination = startDestination.route,
        modifier = modifier
    ) {
        composable(VeloraDestination.HOME.route) {
            HomeScreen(
                uiState = uiState,
                onPlayTrack = { track, pool ->
                    viewModel.playTrack(track, pool)
                    if (track.mediaType == MediaType.VIDEO) {
                        navController.navigateSingleTopTo(VeloraDestination.VIDEO)
                    }
                },
                onPlayAll = { shuffle -> viewModel.playAllMusic(shuffle = shuffle) },
                onPlaySmartPlaylist = { smart, shuffle -> viewModel.playSmartPlaylist(smart, shuffle) },
                onResumeHistory = { entry -> viewModel.resumeFromHistory(entry) },
                onTuneRadio = { station -> viewModel.tuneRadioStation(station) },
                onOpenLibrarySubTab = { subTab ->
                    onSelectLibrarySubTab(subTab)
                    navController.navigateSingleTopTo(VeloraDestination.LIBRARY)
                },
                onOpenNowPlaying = { navController.navigateSingleTopTo(VeloraDestination.PLAYER) },
                onAutoScanMedia = { viewModel.autoScanDeviceMedia() }
            )
        }

        composable(VeloraDestination.LIBRARY.route) {
            LibraryHubScreen(
                uiState = uiState,
                viewModel = viewModel,
                selectedSubTab = librarySubTab,
                onSelectSubTab = onSelectLibrarySubTab,
                onOpenVideoDestination = { navController.navigateSingleTopTo(VeloraDestination.VIDEO) }
            )
        }

        composable(VeloraDestination.MUSIC.route) {
            MusicScreen(
                uiState = uiState,
                onPlayAll = { viewModel.playAllMusic(shuffle = false) },
                onShuffleAll = { viewModel.playAllMusic(shuffle = true) },
                onPlayTrack = { viewModel.playTrack(it, uiState.audioTracks) },
                onAddTrackToQueue = { viewModel.addTrackToQueue(it) },
                onAddTrackToPlaylist = { pid, tid -> viewModel.addTrackToPlaylist(pid, tid) },
                onToggleFavorite = { viewModel.toggleFavorite(it) },
                onAutoScanMedia = { viewModel.autoScanDeviceMedia() }
            )
        }

        composable(VeloraDestination.RADIO.route) {
            RadioScreen(
                uiState = uiState,
                onTuneStation = { viewModel.tuneRadioStation(it) },
                onAddStationToQueue = { viewModel.addRadioStationToQueue(it) },
                onAddCustomStation = { name, url, genre ->
                    viewModel.addCustomRadioStation(name, url, genre)
                }
            )
        }

        composable(VeloraDestination.ARTISTS.route) {
            ArtistScreen(
                uiState = uiState,
                onSelectArtist = { viewModel.selectArtist(it) },
                onPlayArtist = { artist, shuffle -> viewModel.playArtistTracks(artist, shuffle) },
                onPlayTrack = { track ->
                    val artistPool = uiState.artistsMap[track.artist].orEmpty()
                    viewModel.playTrack(track, artistPool)
                },
                onAddTrackToQueue = { viewModel.addTrackToQueue(it) },
                onAddTrackToPlaylist = { pid, tid -> viewModel.addTrackToPlaylist(pid, tid) },
                onToggleFavorite = { viewModel.toggleFavorite(it) }
            )
        }

        composable(VeloraDestination.VIDEO.route) {
            VideoPlayerScreen(
                uiState = uiState,
                playerEngine = viewModel.playerEngine,
                positionState = positionState,
                onPlayVideoTrack = { viewModel.playTrack(it, uiState.videoTracks) },
                onTogglePlayPause = { viewModel.togglePlayPause() },
                onSeekTo = { viewModel.seekTo(it) },
                onSetPlaybackSpeed = { viewModel.setPlaybackSpeed(it) },
                onSetAspectMode = { viewModel.setVideoAspectMode(it) },
                onSetSubtitlesEnabled = { viewModel.setSubtitlesEnabled(it) },
                onSelectAudioTrack = { viewModel.selectAudioTrack(it) },
                onAddTrackToQueue = { viewModel.addTrackToQueue(it) },
                onAddTrackToPlaylist = { pid, tid -> viewModel.addTrackToPlaylist(pid, tid) },
                onToggleFavorite = { viewModel.toggleFavorite(it) },
                onEnterPip = {
                    runCatching {
                        val activity = context as? Activity
                        val params = PictureInPictureParams.Builder()
                            .setAspectRatio(Rational(16, 9))
                            .build()
                        activity?.enterPictureInPictureMode(params)
                    }
                },
                onAutoScanMedia = { viewModel.autoScanDeviceMedia() },
                onSetVolume = { viewModel.setVolume(it) },
                onSetVideoBrightness = { viewModel.setVideoBrightnessLevel(it) },
                onSetSubtitleFontScale = { viewModel.setSubtitleFontScale(it) },
                onSetSubtitleColorTheme = { viewModel.setSubtitleColorTheme(it) },
                onAddVideoBookmark = { viewModel.addTimeMachineBookmark("Cinema Scene") },

            )

/**
 * Mathematically balances and clamps a ColorScheme so all foreground and background
 * pairings strictly adhere to WCAG 2.1 AA luminance ratio (>= 4.5:1) while preserving
 * the user device's Material You dynamic hue.
 */
fun balanceDynamicColorScheme(
    baseScheme: ColorScheme,
    isDark: Boolean
): ColorScheme {
    return if (isDark) {
        val bgInt = HEX_OBSIDIAN
        val primaryInt = (baseScheme.primary.value.toLong() and 0xFFFFFFL).toInt()
        val primaryMeetsAa = PhiDesignMetrics.meetsWcagAa(primaryInt, bgInt, 4.5)
        val balancedPrimary = if (primaryMeetsAa) baseScheme.primary else VeloraBrightSaffron

        baseScheme.copy(
            background = VeloraObsidian,
            onBackground = VeloraPrimaryText,
            surface = VeloraSecondaryBg,
            onSurface = VeloraPrimaryText,
            surfaceVariant = VeloraCardSurface,
            onSurfaceVariant = VeloraSecondaryText,
            primary = balancedPrimary,
            onPrimary = VeloraObsidian,
            outline = VeloraDivider
        )
    } else {
        baseScheme.copy(
            background = VeloraDaylightIvory,
            onBackground = VeloraDaylightInk,
            surface = VeloraDaylightSurface,
            onSurface = VeloraDaylightInk,
            surfaceVariant = VeloraDaylightElevated,
            onSurfaceVariant = VeloraDaylightMutedInk,
            outline = VeloraDaylightBorder
        )
    }
}

@Composable
fun VeloraTheme(
    paletteMode: ThemePaletteMode = ThemePaletteMode.SYSTEM_DYNAMIC,
    darkTheme: Boolean = isSystemInDarkTheme(),
    isPersian: Boolean = false,
    content: @Composable () -> Unit
) {
    val context = LocalContext.current
    val colorScheme = when (paletteMode) {
        ThemePaletteMode.SYSTEM_DYNAMIC -> {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
                val dynamicScheme = if (darkTheme) {
                    dynamicDarkColorScheme(context)
                } else {
                    dynamicLightColorScheme(context)
                }
                balanceDynamicColorScheme(dynamicScheme, isDark = darkTheme)
            } else {
                if (darkTheme) VeloraSaffronColorScheme else VeloraDaylightColorScheme
            }
        }
        ThemePaletteMode.VELORA_SAFFRON -> VeloraSaffronColorScheme
        ThemePaletteMode.VELORA_DAYLIGHT -> VeloraDaylightColorScheme
        ThemePaletteMode.HIGH_CONTRAST_MONOCHROME -> VeloraHighContrastColorScheme
    }

    val semanticColors = when (paletteMode) {
        ThemePaletteMode.VELORA_SAFFRON -> VeloraSaffronColors
        ThemePaletteMode.VELORA_DAYLIGHT -> VeloraDaylightColors
        ThemePaletteMode.HIGH_CONTRAST_MONOCHROME -> VeloraHighContrastColors
        ThemePaletteMode.SYSTEM_DYNAMIC -> {
            val base = if (darkTheme) VeloraSaffronColors else VeloraDaylightColors
            base.copy(
                accent = colorScheme.primary,
                accentContainer = colorScheme.primaryContainer,
                onAccentContainer = colorScheme.onPrimaryContainer
            )
        }
    }

    CompositionLocalProvider(LocalVeloraColors provides semanticColors) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = veloraTypography(isPersian = isPersian),
            shapes = VeloraShapes,
            content = content
        )
    }
}

@Composable
internal fun FullscreenVideoDialog(
    context: Context,
    playerEngine: PlayerEngine?,
    uiState: FazoraUiState,
    positionState: PlaybackPositionState,
    controlsLocked: Boolean,
    gestureHudLabel: String?,
    activeSubtitleText: String,
    onToggleControls: () -> Unit,
    onToggleLock: () -> Unit,
    onExitFullscreen: () -> Unit,
    onSeekTo: (Long) -> Unit,
    onTogglePlayPause: () -> Unit,
    onSetBrightness: (Float) -> Unit,
    onSetVolume: (Float) -> Unit,
    onHudNotice: (String) -> Unit
) {
    val engine = uiState.engineState
    val nativeRatio = when (engine.videoAspectMode) {
        VideoAspectMode.GOLDEN_PHI -> PhiTokens.PHI
        VideoAspectMode.SIXTEEN_NINE, VideoAspectMode.FIT, VideoAspectMode.FILL -> 16f / 9f
    }

    DisposableEffect(nativeRatio) {
        val activity = context as? Activity
        val prevOrientation = activity?.requestedOrientation ?: ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED
        val window = activity?.window
        val insetsController = if (window != null) {
            WindowCompat.getInsetsController(window, window.decorView)
        } else {
            null
        }
        insetsController?.systemBarsBehavior =
            WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        insetsController?.hide(WindowInsetsCompat.Type.systemBars())

        if (FullscreenPolicy.shouldLockLandscape(nativeRatio)) {
            activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE
        }
        onDispose {
            insetsController?.show(WindowInsetsCompat.Type.systemBars())
            activity?.requestedOrientation = prevOrientation
        }
    }

    Dialog(
        onDismissRequest = onExitFullscreen,
        properties = DialogProperties(usePlatformDefaultWidth = false)
    ) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(MaterialTheme.colorScheme.background)
                .veloraCinemaGestures(
                    controlsLocked = controlsLocked,
                    positionMs = positionState.positionMs,
                    durationMs = positionState.durationMs,
                    currentBrightness = uiState.videoCinemaSettings.brightnessLevel,
                    currentVolume = engine.volume,
                    onToggleControls = onToggleControls,
                    onInteraction = {},
                    onSeekTo = onSeekTo,
                    onTogglePlayPause = onTogglePlayPause,
                    onSetBrightness = onSetBrightness,
                    onSetVolume = onSetVolume,
                    onHudNotice = onHudNotice
                )
                .testTag("fullscreen_video_dialog")
        ) {
            VeloraVideoSurface(
                playerEngine = playerEngine,
                videoAspectMode = engine.videoAspectMode,
                surfaceTestTag = "fullscreen_video_surface",
                modifier = Modifier.fillMaxSize()
            )

            if (engine.subtitlesEnabled && activeSubtitleText.isNotBlank()) {
                VideoSubtitleOverlay(
                    text = activeSubtitleText,
                    cinemaSettings = uiState.videoCinemaSettings,
                    modifier = Modifier
                        .align(Alignment.BottomCenter)
                        .padding(PhiTokens.Space34)
                )
            }

            Row(
                horizontalArrangement = Arrangement.SpaceBetween,
                modifier = Modifier
                    .fillMaxWidth()
                    .align(Alignment.TopCenter)
                    .padding(PhiTokens.Space21)
            ) {
                IconButton(
                    onClick = onToggleLock,
                    modifier = Modifier.minimumInteractiveComponentSize()
                ) {
                    Icon(
                        imageVector = if (controlsLocked) Icons.Default.Lock else Icons.Default.LockOpen,
                        contentDescription = "Toggle Lock",
                        tint = MaterialTheme.colorScheme.primary
                    )

                        onClick = onExitFullscreen,

                            tint = MaterialTheme.colorScheme.primary

package com.example.ui.components

import androidx.compose.ui.Modifier
import androidx.compose.ui.input.key.Key
import androidx.compose.ui.input.key.KeyEvent
import androidx.compose.ui.input.key.KeyEventType
import androidx.compose.ui.input.key.isAltPressed
import androidx.compose.ui.input.key.isCtrlPressed
import androidx.compose.ui.input.key.isMetaPressed
import androidx.compose.ui.input.key.key
import androidx.compose.ui.input.key.onKeyEvent
import androidx.compose.ui.input.key.type

enum class VeloraKeyboardShortcutAction {
    TOGGLE_PLAY_PAUSE,
    SEEK_FORWARD_5S,
    SEEK_BACKWARD_5S,
    VOLUME_UP,
    VOLUME_DOWN,
    TOGGLE_MUTE,
    TOGGLE_VIDEO_FULLSCREEN
}

/**
 * Pure key-event mapper for hardware keyboard playback shortcuts so behavior is deterministic
 * and unit-testable without requiring a physical Android device or Desktop JVM window.
 */
fun resolveVeloraKeyboardShortcut(event: KeyEvent): VeloraKeyboardShortcutAction? {
    if (event.type != KeyEventType.KeyDown) return null
    if (event.isCtrlPressed || event.isAltPressed || event.isMetaPressed) return null
    return when (event.key) {
        Key.Spacebar -> VeloraKeyboardShortcutAction.TOGGLE_PLAY_PAUSE
        Key.DirectionRight -> VeloraKeyboardShortcutAction.SEEK_FORWARD_5S
        Key.DirectionLeft -> VeloraKeyboardShortcutAction.SEEK_BACKWARD_5S
        Key.DirectionUp -> VeloraKeyboardShortcutAction.VOLUME_UP
        Key.DirectionDown -> VeloraKeyboardShortcutAction.VOLUME_DOWN
        Key.M -> VeloraKeyboardShortcutAction.TOGGLE_MUTE
        Key.F -> VeloraKeyboardShortcutAction.TOGGLE_VIDEO_FULLSCREEN
        else -> null
    }
}

fun Modifier.veloraKeyboardShortcuts(
    onAction: (VeloraKeyboardShortcutAction) -> Unit
): Modifier = onKeyEvent { event ->
    val action = resolveVeloraKeyboardShortcut(event)
    if (action != null) {
        onAction(action)
        true
    } else {
        false
    }
}

@Composable
fun VeloraSeekBar(
    positionMs: Long,
    durationMs: Long,
    onSeekTo: (Long) -> Unit,
    modifier: Modifier = Modifier,
    sliderTestTag: String = "player_seek_slider"
) {
    val safeDuration = durationMs.coerceAtLeast(1L)
    var dragFraction by remember { mutableStateOf<Float?>(null) }
    val effectiveMs = dragFraction?.let { (it * safeDuration).toLong() } ?: positionMs.coerceIn(0L, safeDuration)
    val fraction = (effectiveMs.toFloat() / safeDuration.toFloat()).coerceIn(0f, 1f)

    Column(modifier = modifier.fillMaxWidth()) {
        Slider(
            value = fraction,
            onValueChange = { dragFraction = it },
            onValueChangeFinished = {
                val target = ((dragFraction ?: fraction) * safeDuration).toLong()
                dragFraction = null
                onSeekTo(PlaybackQueue.clampSeekPosition(target, safeDuration))
            },
            colors = SliderDefaults.colors(
                thumbColor = MaterialTheme.colorScheme.primary,
                activeTrackColor = MaterialTheme.colorScheme.primary,
                inactiveTrackColor = MaterialTheme.colorScheme.surfaceVariant
            ),
            modifier = Modifier
                .fillMaxWidth()
                .testTag(sliderTestTag)
        )
        Row(
            horizontalArrangement = Arrangement.SpaceBetween,
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(
                text = Track.formatDurationMs(effectiveMs),
                style = MaterialTheme.typography.labelMedium,
                color = MaterialTheme.colorScheme.primary,
                modifier = Modifier.testTag("elapsed_time_label")
            )
            Text(
                text = Track.formatDurationMs(durationMs),
                style = MaterialTheme.typography.labelMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                modifier = Modifier.testTag("total_duration_label")
            )
        }
    }
}

/**
 * 10-Band Goertzel Real-Time Spectrum & Live Biquad Transfer Curve Visualizer.
 * Uses only theme color tokens and PhiTokens sizing.
 */
@Composable
fun SpectrumAnalyzerCanvas(
    spectrumBands: List<Float>,
    eqBandGainsDb: List<Float>,
    isPlaying: Boolean,
    visualizerMode: VisualizerMode = VisualizerMode.SPECTRUM_BARS,
    modifier: Modifier = Modifier
) {
    val primaryColor = MaterialTheme.colorScheme.primary
    val secondaryColor = MaterialTheme.colorScheme.secondary
    val tertiaryColor = MaterialTheme.colorScheme.tertiary
    val surfaceColor = MaterialTheme.colorScheme.surface
    val outlineColor = MaterialTheme.colorScheme.outline

    val transition = rememberInfiniteTransition(label = "spectrum_phase")
    val phase by transition.animateFloat(
        initialValue = 0f,
        targetValue = (2.0 * PI).toFloat(),
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = 2200, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "wave_phase"
    )

    val transferCurveDb = remember(eqBandGainsDb) {
        BiquadEqDsp.evaluateCascadeMagnitudeResponseDb(
            sampleRateHz = 48_000,
            gainsDb = eqBandGainsDb,
            pointCount = 32
        )
    }

    Canvas(
        modifier = modifier
            .fillMaxWidth()
            .height(PhiTokens.SpectrumHeight)
            .clip(RoundedCornerShape(PhiTokens.CornerRadius13))
            .background(surfaceColor)
            .border(
                PhiTokens.BorderHairline,
                outlineColor,
                RoundedCornerShape(PhiTokens.CornerRadius13)
            )
            .testTag("spectrum_analyzer_canvas")
    ) {
        val width = size.width
        val height = size.height
        if (width <= 0f || height <= 0f) return@Canvas

        drawLine(
            color = outlineColor,
            start = Offset(0f, height * 0.5f),
            end = Offset(width, height * 0.5f),
            strokeWidth = 1f
        )

        val bands = if (spectrumBands.isNotEmpty()) spectrumBands else List(10) { 0.12f }
        when (visualizerMode) {
            VisualizerMode.WAVEFORM_OSCILLOSCOPE -> {
                val wavePath = Path()
                val steps = 48
                for (i in 0..steps) {
                    val x = (i.toFloat() / steps.toFloat()) * width
                    val bandIdx = ((i.toFloat() / steps) * (bands.size - 1)).toInt().coerceIn(0, bands.lastIndex)
                    val amp = if (isPlaying) bands[bandIdx].coerceIn(0.08f, 1f) else 0.08f
                    val y = height * 0.5f + sin(phase + i * 0.35f) * (height * 0.38f * amp)
                    if (i == 0) wavePath.moveTo(x, y) else wavePath.lineTo(x, y)
                }
                drawPath(
                    path = wavePath,
                    color = primaryColor,
                    style = Stroke(width = 3f)
                )
            }
            VisualizerMode.DUAL_STUDIO, VisualizerMode.SPECTRUM_BARS -> {
                val count = bands.size.coerceAtLeast(1)
                val slotWidth = width / count.toFloat()
                val barWidth = slotWidth * PhiTokens.PRIMARY_WEIGHT
                bands.forEachIndexed { idx, rawLevel ->
                    val animatedFloor = if (isPlaying) {
                        (0.14f + 0.06f * sin(phase + idx * 0.6f)).coerceAtLeast(0.08f)
                    } else {
                        0.06f
                    }
                    val level = rawLevel.coerceAtLeast(animatedFloor).coerceIn(0.05f, 1f)
                    val barHeight = height * 0.82f * level
                    val left = idx * slotWidth + (slotWidth - barWidth) * 0.5f
                    val top = height - barHeight
                    drawRoundRect(
                        brush = Brush.verticalGradient(
                            colors = listOf(tertiaryColor, primaryColor, secondaryColor.copy(alpha = 0.45f)),
                            startY = top,
                            endY = height
                        ),
                        topLeft = Offset(left, top),
                        size = Size(barWidth, barHeight),
                        cornerRadius = CornerRadius(6f, 6f)
                    )
                }
            }
        }

        if (transferCurveDb.size >= 2) {
            val eqPath = Path()
            transferCurveDb.forEachIndexed { i, db ->
                val x = (i.toFloat() / (transferCurveDb.size - 1).toFloat()) * width
                val normalized = ((db + 15f) / 30f).coerceIn(0f, 1f)
                val y = height * (1f - normalized)
                if (i == 0) eqPath.moveTo(x, y) else eqPath.lineTo(x, y)
            }
            drawPath(
                path = eqPath,
                color = secondaryColor,
                style = Stroke(width = 2.5f)
            )
        }
    }
}

/**
 * Platform-specific video rendering surface boundary (`androidMain` implementation).
 * Keeps `ExoPlayer` and `PlayerView` encapsulated inside a single UI adapter component
 * so screen composables depend only on the multiplatform-ready [PlayerEngine] contract (F10, F16).
 */
@OptIn(UnstableApi::class)
@Composable
fun VeloraVideoSurface(
    playerEngine: PlayerEngine?,
    videoAspectMode: VideoAspectMode,
    surfaceTestTag: String,
    modifier: Modifier = Modifier
) {
    val exoPlayer = (playerEngine as? Media3PlayerEngine)?.exoPlayer ?: return
    AndroidView(
        factory = { ctx ->
            val playerView = LayoutInflater.from(ctx).inflate(
                R.layout.velora_video_player_view,
                null,
                false
            ) as PlayerView
            playerView.apply {
                setEnableComposeSurfaceSyncWorkaround(true)
                setShutterBackgroundColor(android.graphics.Color.TRANSPARENT)
                useController = false
                keepScreenOn = true
                player = exoPlayer
                onResume()
            }
        },
        update = { view ->
            if (view.player !== exoPlayer) {
                view.player = exoPlayer
            }
            view.onResume()
            view.resizeMode = when (videoAspectMode) {
                VideoAspectMode.FIT -> AspectRatioFrameLayout.RESIZE_MODE_FIT
                VideoAspectMode.FILL -> AspectRatioFrameLayout.RESIZE_MODE_ZOOM
                VideoAspectMode.GOLDEN_PHI -> AspectRatioFrameLayout.RESIZE_MODE_FILL
                VideoAspectMode.SIXTEEN_NINE -> AspectRatioFrameLayout.RESIZE_MODE_FIT
            }
        },
        onRelease = { view -> view.player = null },
        modifier = modifier.testTag(surfaceTestTag)
    )
}

/**
 * Shared cinema gesture modifier for both inline and fullscreen video surfaces (F16).
 */
fun Modifier.veloraCinemaGestures(
    controlsLocked: Boolean,
    positionMs: Long,
    durationMs: Long,
    currentBrightness: Float,
    currentVolume: Float,
    onToggleControls: () -> Unit,
    onInteraction: () -> Unit,
    onSeekTo: (Long) -> Unit,
    onTogglePlayPause: () -> Unit,
    onSetBrightness: (Float) -> Unit,
    onSetVolume: (Float) -> Unit,
    onHudNotice: (String) -> Unit
): Modifier = this
    .pointerInput(controlsLocked, positionMs, durationMs) {
        detectTapGestures(
            onDoubleTap = { offset ->
                if (!controlsLocked) {
                    onInteraction()
                    val res = VeloraGestureEngine.resolveDoubleTapGesture(
                        tapX = offset.x,
                        surfaceWidth = size.width.toFloat(),
                        currentPositionMs = positionMs,
                        durationMs = durationMs
                    )
                    when (res.action) {
                        GestureActionType.SEEK_BACKWARD,
                        GestureActionType.SEEK_FORWARD -> {
                            onSeekTo(res.targetPositionMs)
                            onHudNotice(res.hudLabel)
                        }
                        else -> {
                            onTogglePlayPause()
                            onHudNotice(res.hudLabel)
                        }
                    }
                }
            },
            onTap = {
                onToggleControls()
                onInteraction()
            }
        )
    }
    .pointerInput(controlsLocked, currentBrightness, currentVolume) {
        var dragStartX = 0f
        detectVerticalDragGestures(
            onDragStart = { offset -> dragStartX = offset.x },
            onVerticalDrag = { change, dragAmount ->
                if (!controlsLocked) {
                    change.consume()
                    val swipe = VeloraGestureEngine.resolveVerticalSwipeGesture(
                        startX = dragStartX,
                        surfaceWidth = size.width.toFloat(),
                        surfaceHeight = size.height.toFloat(),
                        dragDeltaY = dragAmount,
                        currentBrightness = currentBrightness,
                        currentVolume = currentVolume
                    )
                    if (swipe.action == GestureActionType.ADJUST_BRIGHTNESS) {
                        onSetBrightness(swipe.newValue)
                    } else {
                        onSetVolume(swipe.newValue)
                    }
                    onHudNotice(swipe.hudLabel)
                }
            }
        )
    }
